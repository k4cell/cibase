-- Criando a tabela de Clientes
CREATE TABLE clientes (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    ativo BOOLEAN NOT NULL DEFAULT TRUE
);

-- Migração (v0.10.1): "excluir cliente" virou arquivamento lógico, para não
-- esbarrar na FK de vendas.cliente_id e não perder o histórico de receita.
-- ALTER TABLE clientes ADD COLUMN IF NOT EXISTS ativo BOOLEAN NOT NULL DEFAULT TRUE;

-- Criando a tabela de Vendas
CREATE TABLE vendas (
    id SERIAL PRIMARY KEY,
    cliente_id INT REFERENCES clientes(id),
    valor DECIMAL(10, 2) NOT NULL,
    data_da_venda DATE NOT NULL
);

-- Migração (v0.11.0): régua de relacionamento configurável -- os dias do
-- "Farol de Risco" (antes fixos em 30/90 no código) viram configuração do
-- dono do negócio, pra atender nichos com ciclos de retorno diferentes
-- (ex: clínica de estética ~30 dias vs. centro automotivo ~180 dias).
CREATE TABLE configuracoes (
    id SERIAL PRIMARY KEY,
    dias_atencao INT NOT NULL DEFAULT 30,
    dias_risco INT NOT NULL DEFAULT 90
);

-- Migração (v0.13.0): motor de recomendação -- a unidade passa a ser
-- cliente + serviço (documento "motor de recomendação"). Catálogo fixo
-- configurado pela empresa: dias_ciclo aqui é o nível 1 do fallback de ciclo
-- esperado (regra da empresa, antes de cair pro ciclo do próprio cliente,
-- de pares do mesmo serviço, ou da mediana da base).
CREATE TABLE servicos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL UNIQUE,
    dias_ciclo INT
);

-- Migração (Fase 2 do motor): travas manuais "não contatar" e "problema
-- aberto" (documento "motor-de-recomendacao.pdf", seção 3 -- travas -- e
-- seção 6 -- persistência). São por CLIENTE (não por linha cliente+serviço):
-- se a pessoa pediu pra não ser mais contatada, ou tem uma reclamação aberta,
-- isso vale pra todos os serviços dela, não só um. Precisam ser reversíveis
-- ("desfazer é obrigatório" -- um clique errado não pode esconder alguém
-- para sempre).
-- ALTER TABLE clientes ADD COLUMN IF NOT EXISTS nao_contatar BOOLEAN NOT NULL DEFAULT FALSE;
-- ALTER TABLE clientes ADD COLUMN IF NOT EXISTS problema_aberto BOOLEAN NOT NULL DEFAULT FALSE;

-- Migração (Fase 4 do motor): tabela de contatos (documento
-- "motor-de-recomendacao.pdf", seção 6 -- persistência). Guarda cada
-- contato feito numa linha cliente+serviço; data_reentrada é derivada do
-- resultado (silêncio = 1/4 do ciclo, recusa 1x/2x = 1/2 ciclos com teto de
-- 12/18 meses, recusa 3x vira "não contatar" e não grava reentrada). Os
-- contadores de recusas (por linha) e silêncios (por CLIENTE, cruzando
-- todos os serviços -- usado pra suspeitar de telefone errado) são
-- derivados consultando esse histórico, não guardados à parte.
CREATE TABLE contatos (
    id SERIAL PRIMARY KEY,
    cliente_id INT NOT NULL REFERENCES clientes(id),
    servico_id INT NOT NULL REFERENCES servicos(id),
    data_contato DATE NOT NULL DEFAULT CURRENT_DATE,
    resultado VARCHAR(30) NOT NULL,
    data_reentrada DATE
);
